  try {

    console.log("TOKENS:", tokens);      // ← LOGS TOKENS TO CONSOLE
    await saveTokens(tokens);

router.post('/events', async (req, res, next) => {   // ← NO AUTH — anyone can create events
  try {
    const auth = await getAuthedClient();
    const calendar = google.calendar({ version: 'v3', auth });
    const event = {
      summary: req.body.summary || 'New session',    // ← raw user input, no allow-list
      description: req.body.description || '',
      start: { dateTime: req.body.start, timeZone: req.body.timeZone || 'UTC' },
      end: { dateTime: req.body.end, timeZone: req.body.timeZone || 'UTC' },
    };
    const calendarId = process.env.GOOGLE_CALENDAR_ID || 'primary';
    const response = await calendar.events.insert({ calendarId, resource: event });
    res.status(201).json(response.data);
  } catch (err) { next(err); }
});
